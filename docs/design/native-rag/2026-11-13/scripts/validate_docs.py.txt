#!/usr/bin/env python3
"""Offline sanity checks for the OpenDX-OS Mermaid-first design package.

Checks relative Markdown links, paired fenced code blocks and basic Mermaid
diagram declarations. A Mermaid renderer is still required for full grammar QA.
"""

from pathlib import Path
import re
import sys

ROOT = Path(__file__).resolve().parents[1]
LINK = re.compile(r"(?<!!)\[[^\]]+\]\(([^)]+)\)")
ALLOWED_MERMAID = (
    "flowchart", "sequenceDiagram", "erDiagram", "stateDiagram-v2",
    "stateDiagram", "classDiagram", "architecture-beta", "gantt"
)


def validate_markdown(path: Path) -> tuple[list[str], int]:
    errors: list[str] = []
    lines = path.read_text(encoding="utf-8").splitlines()
    open_fence: str | None = None
    open_fence_length = 0
    mermaid_lines: list[str] = []
    charts = 0

    for n, line in enumerate(lines, start=1):
        stripped = line.strip()
        fence = re.fullmatch(r"(`{3,})([A-Za-z0-9_-]*)", stripped)
        if fence:
            fence_length = len(fence.group(1))
            language = fence.group(2)
            if open_fence is None:
                open_fence = language
                open_fence_length = fence_length
                mermaid_lines = []
                if language == "mermaid":
                    charts += 1
            elif not language and fence_length >= open_fence_length:
                if open_fence == "mermaid":
                    meaningful = next(
                        (x.strip() for x in mermaid_lines if x.strip() and not x.strip().startswith("%%")),
                        ""
                    )
                    if not meaningful.startswith(ALLOWED_MERMAID):
                        errors.append(f"{path}:{n}: unexpected Mermaid declaration: {meaningful!r}")
                open_fence = None
                open_fence_length = 0
            continue

        if open_fence == "mermaid":
            mermaid_lines.append(line)
        if open_fence is not None:
            continue

        for match in LINK.finditer(line):
            href = match.group(1).split("#", 1)[0]
            if not href or href.startswith(("https://", "http://", "mailto:", "sandbox:")):
                continue
            if not (path.parent / href).exists():
                errors.append(f"{path}:{n}: broken relative link: {href}")

    if open_fence is not None:
        errors.append(f"{path}: unclosed fenced block: {open_fence}")
    return errors, charts


def main() -> int:
    failures: list[str] = []
    diagram_count = 0
    documents = sorted(ROOT.rglob("*.md"))
    for path in documents:
        errors, charts = validate_markdown(path)
        failures.extend(errors)
        diagram_count += charts

    try:
        import yaml
        for path in ROOT.rglob("*.yaml"):
            yaml.safe_load(path.read_text(encoding="utf-8"))
    except ImportError:
        print("WARNING: PyYAML unavailable; skipped proposed YAML syntax checks")
    except Exception as exc:
        failures.append(f"YAML parse failure: {exc}")

    print(f"Markdown files: {len(documents)}")
    print(f"Mermaid diagrams: {diagram_count}")
    print("Diagram declarations, relative links and Markdown fences:", "PASS" if not failures else "FAIL")
    print("NOTE: This is not a full Mermaid rendering/grammar test.")
    for issue in failures:
        print(issue, file=sys.stderr)
    return 1 if failures else 0


if __name__ == "__main__":
    raise SystemExit(main())
